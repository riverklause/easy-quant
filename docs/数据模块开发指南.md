```python
async def analyze_fundamental_data():
    """批量分析基本面数据"""
    
    stocks = ["AAPL", "GOOG", "MSFT", "AMZN"]
    fundamental_data = {}
    
    for stock in stocks:
        try:
            # 获取财务指标
            indicators = await data_manager.get_financial_indicators(
                symbol=f"US.{stock}",
                data_source="yfinance",
                period="annual"
            )
            
            # 提取关键指标
            key_metrics = {
                'market_cap': indicators.loc['market_cap', 'value'] if 'market_cap' in indicators.index else None,
                'pe_ratio': indicators.loc['pe_ratio', 'value'] if 'pe_ratio' in indicators.index else None,
                'pb_ratio': indicators.loc['pb_ratio', 'value'] if 'pb_ratio' in indicators.index else None,
                'roe': indicators.loc['roe', 'value'] if 'roe' in indicators.index else None
            }
            
            fundamental_data[stock] = key_metrics
            print(f"{stock}: 市值={key_metrics['market_cap']:.2f}, PE={key_metrics['pe_ratio']:.2f}")
            
        except Exception as e:
            print(f"获取{stock}基本面数据失败: {e}")
    
    # 转换为DataFrame进行分析
    df_fundamental = pd.DataFrame(fundamental_data).T
    print("\n基本面数据汇总:")
    print(df_fundamental)
    
    return df_fundamental

asyncio.run(analyze_fundamental_data())
```

### 实时市场监控

```python
import time

async def monitor_market():
    """实时市场监控"""
    
    symbols = ["US.AAPL", "US.GOOG", "US.MSFT"]
    
    print("开始市场监控...")
    print("股票代码\t当前价格\t涨跌幅\t成交量")
    
    for i in range(5):  # 监控5次
        snapshot = await data_manager.get_market_snapshot(
            symbols=symbols,
            data_source="yfinance"
        )
        
        for symbol in symbols:
            if symbol in snapshot.index:
                data = snapshot.loc[symbol]
                price = data.get('last_price', 'N/A')
                prev_close = data.get('prev_close_price', price)
                
                if price != 'N/A' and prev_close != 'N/A':
                    change_pct = (price - prev_close) / prev_close * 100
                    volume = data.get('volume', 0)
                    
                    print(f"{symbol}\t{price:.2f}\t{change_pct:+.2f}%\t{volume:,}")
                else:
                    print(f"{symbol}\t数据暂不可用")
        
        print("-" * 50)
        
        if i < 4:  # 不是最后一次循环
            time.sleep(60)  # 等待1分钟
    
    print("市场监控结束")

asyncio.run(monitor_market())
```

| 缓存类型 | 说明 | 最大容量 | 按股票分组 |
|---------|------|---------|-----------|
| quote | 实时报价 | 1000条/股 | 是 |
| order_book | 实时摆盘 | 1000条/股 | 是 |

### 实时数据回调

实时数据通过 DataFrame 格式传递，以便后续指标计算模块直接使用向量化计算。

```python
def on_realtime_data(data):
    """
    实时数据回调函数
    
    注意：data 参数现在是 pd.DataFrame 类型（单行），而非 dict
    可以直接传入指标计算模块进行向量化计算
    """
    import pandas as pd
    
    if data is None or data.empty:
        return
    
    data_type = data['data_type'].iloc[0] if 'data_type' in data.columns else None
    symbol = data['code'].iloc[0] if 'code' in data.columns else None
    
    if data_type == 'quote':
        price = float(data['last_price'].iloc[0]) if 'last_price' in data.columns else None
        volume = int(data['volume'].iloc[0]) if 'volume' in data.columns else None
        print(f"实时报价: {symbol} 价格={price}, 成交量={volume}")
    
    elif data_type == 'kline':
        open_price = float(data['open'].iloc[0]) if 'open' in data.columns else None
        close_price = float(data['close'].iloc[0]) if 'close' in data.columns else None
        print(f"K线数据: {symbol} 开盘={open_price}, 收盘={close_price}")
        
        # DataFrame 可直接用于指标计算
        # from indicators import MovingAverage
        # ma = MovingAverage(data, period=5)
    
    elif data_type == 'order_book':
        bid_prices = data['bid_prices'].iloc[0] if 'bid_prices' in data.columns else []
        ask_prices = data['ask_prices'].iloc[0] if 'ask_prices' in data.columns else []
        print(f"摆盘数据: {symbol} 买价={bid_prices[:3]}, 卖价={ask_prices[:3]}")

async def subscribe_realtime_data():
    """订阅实时数据"""
    
    # 注册回调函数（DataManager 内置回调处理）
    data_manager.register_callback('futu_RT', data_manager.on_realtime_data)
    
    # 订阅实时数据
    symbols = ['HK.00700', 'HK.00941']
    data_types = ['kline', 'quote', 'order_book']
    
    success = data_manager.subscribe_realtime_data(
        data_source='futu_RT',
        symbols=symbols,
        data_types=data_types
    )
    
    if success:
        print(f"成功订阅实时数据: {symbols}")
        
        # 保持运行一段时间以接收数据
        print("开始接收实时数据，按Ctrl+C停止...")
        try:
            await asyncio.sleep(300)  # 运行5分钟
        except KeyboardInterrupt:
            print("\n停止接收实时数据")
    else:
        print("订阅实时数据失败")
```

## 数据存储管理

### 数据目录结构

Easy-Quant 的数据存储采用以下目录结构：

```
data/
├── StockList.parquet           # 关注股票列表
├── StockBasicInfo.parquet      # 市场股票全列表
├── StockBasicFiltered.parquet # 粗过滤后的活跃股票列表
├── history_data/              # 用于策略分析的历史数据
│   └── {symbol}/
├── screener/                  # 用于股票过滤的数据
├── realtime/                  # 实时数据
│   ├── kline/                # 实时K线数据
│   ├── tick/                 # 实时成交数据
│   ├── quote/                # 实时报价数据
│   └── order_book/           # 实时摆盘数据
├── sources/                   # 数据源客户端
├── storage/                   # 数据存储方法
├── rehab/                     # 复权因子数据
├── snapshot/                 # 快照数据
└── temp/                     # 其他临时数据
```

```python
async def optimized_batch_operations():

            successful_results.append(result)

    return successful_results

asyncio.run(optimized_batch_operations())
```

### 数据缓存优化

```python
from utils.cache import DataCache

class OptimizedDataManager:
    """带缓存优化的数据管理器"""
    
    def __init__(self):
        self.data_manager = DataManager()
        self.cache = DataCache(max_size=1000, ttl=3600)  # 缓存1小时
    
    async def get_historical_data_with_cache(self, symbol, start_date, end_date, **kwargs):
        """带缓存的历史数据获取"""
        
        cache_key = f"historical_{symbol}_{start_date}_{end_date}"
        
        # 检查缓存
        cached_data = self.cache.get(cache_key)
        if cached_data is not None:
            print(f"从缓存获取数据: {symbol}")
            return cached_data
        
        # 从数据源获取
        data = await self.data_manager.get_historical_data(
            symbol=symbol,
            start_date=start_date,
            end_date=end_date,
            **kwargs
        )
        
        # 存入缓存
        self.cache.set(cache_key, data)
        print(f"数据已缓存: {symbol}")
        
        return data
```

## 常见问题解答

### Q1: 数据获取失败怎么办？

**A:** 检查以下方面：
1. 数据源连接状态：`data_manager.health_check()`
2. 股票代码格式是否正确
3. 网络连接是否正常
4. 数据源服务是否运行（如Futu OpenD）

### Q2: 如何提高数据获取速度？

**A:** 使用以下优化方法：
1. 批量获取多个股票数据
2. 使用缓存机制减少重复请求
3. 并行处理多个数据请求
4. 选择合适的数据源
5. 使用asyncio.gather并行请求

### Q3: 如何处理数据过期？

**A:** 使用数据过期判断功能：
1. 文件时间判断：基于文件修改时间
2. 数据内容判断：基于数据最新时间戳
3. 混合判断：结合两种方式

```python
# 使用混合过期判断
data = storage_manager.load_data(
    symbol='US.AAPL',
    data_type='historical',
    max_age_hours=24,
    expiry_check_method='hybrid'  # 混合判断
)
```

### Q4: 如何保存实时数据？

**A:** 使用LocalStorageManager进行数据缓冲和保存：
```python
# 每隔一定数量保存一次
if len(buffer) >= 10:
    storage_manager.save_data(data=buffer, ...)
```
