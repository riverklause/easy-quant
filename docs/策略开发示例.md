### 策略基类使用

```python
from strategy.base_strategy import BaseStrategy
import pandas as pd
import numpy as np

class SimpleMovingAverageStrategy(BaseStrategy):
    """简单移动平均线策略"""
    
    def __init__(self, fast_period=10, slow_period=30, name="SMA_Strategy"):
        super().__init__(name=name)
        self.fast_period = fast_period
        self.slow_period = slow_period
        self.position = 0  # 持仓状态：0-空仓，1-多仓，-1-空头
    
    async def on_data(self, data):
        """数据处理回调"""
        if len(data) < self.slow_period:
            return  # 数据不足，不生成信号
        
        # 计算移动平均线
        fast_sma = data['close'].rolling(self.fast_period).mean()
        slow_sma = data['close'].rolling(self.slow_period).mean()
        
        current_fast = fast_sma.iloc[-1]
        current_slow = slow_sma.iloc[-1]
        previous_fast = fast_sma.iloc[-2]
        previous_slow = slow_sma.iloc[-2]
        
        current_price = data['close'].iloc[-1]
        
        # 生成交易信号
        if current_fast > current_slow and previous_fast <= previous_slow:
            # 金叉信号：买入
            if self.position != 1:
                self.generate_signal(
                    symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                    signal_type='BUY',
                    price=current_price,
                    quantity=100,  # 固定数量
                    reason=f"SMA金叉：快线{current_fast:.2f} > 慢线{current_slow:.2f}"
                )
                self.position = 1
        
        elif current_fast < current_slow and previous_fast >= previous_slow:
            # 死叉信号：卖出
            if self.position != -1:
                self.generate_signal(
                    symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                    signal_type='SELL',
                    price=current_price,
                    quantity=100,
                    reason=f"SMA死叉：快线{current_fast:.2f} < 慢线{current_slow:.2f}"
                )
                self.position = -1
    
    def on_signal(self, signal):
        """信号处理回调"""
        print(f"策略 {self.name} 生成信号: {signal}")
        # 这里可以添加交易执行逻辑

# 使用示例
sma_strategy = SimpleMovingAverageStrategy(fast_period=10, slow_period=30)
```

```python
from indicators.technical.momentum.rsi import RSI

class RSIStrategy(BaseStrategy):
    """RSI超买超卖策略"""
    
    def __init__(self, rsi_period=14, oversold=30, overbought=70, name="RSI_Strategy"):
        super().__init__(name=name)
        self.rsi_period = rsi_period
        self.oversold = oversold
        self.overbought = overbought
        self.rsi_indicator = RSI(period=rsi_period)
        self.position = 0
    
    async def on_data(self, data):
        """数据处理"""
        if len(data) < self.rsi_period + 1:
            return
        
        # 计算RSI
        rsi_result = self.rsi_indicator.calculate(data['close'])
        current_rsi = rsi_result.iloc[-1]
        
        current_price = data['close'].iloc[-1]
        
        # 生成交易信号
        if current_rsi < self.oversold and self.position <= 0:
            # RSI超卖，买入信号
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='BUY',
                price=current_price,
                quantity=100,
                reason=f"RSI超卖: {current_rsi:.2f} < {self.oversold}"
            )
            self.position = 1
        
        elif current_rsi > self.overbought and self.position >= 0:
            # RSI超买，卖出信号
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='SELL',
                price=current_price,
                quantity=100,
                reason=f"RSI超买: {current_rsi:.2f} > {self.overbought}"
            )
            self.position = -1
    
    def get_parameters(self):
        """获取策略参数"""
        return {
            'rsi_period': self.rsi_period,
            'oversold': self.oversold,
            'overbought': self.overbought
        }

# 使用示例
rsi_strategy = RSIStrategy(rsi_period=14, oversold=30, overbought=70)
```

```python
from indicators.technical.volatility.bollinger_bands import BollingerBands

class BollingerBandsStrategy(BaseStrategy):
    """布林带均值回归策略"""
    
    def __init__(self, period=20, std_dev=2, name="Bollinger_Strategy"):
        super().__init__(name=name)
        self.period = period
        self.std_dev = std_dev
        self.bb_indicator = BollingerBands(period=period, std_dev=std_dev)
        self.position = 0
    
    async def on_data(self, data):
        """数据处理"""
        if len(data) < self.period:
            return
        
        # 计算布林带
        bb_result = self.bb_indicator.calculate(data['close'])
        
        current_price = data['close'].iloc[-1]
        upper_band = bb_result['upper_band'].iloc[-1]
        lower_band = bb_result['lower_band'].iloc[-1]
        middle_band = bb_result['middle_band'].iloc[-1]
        
        # 生成交易信号
        if current_price <= lower_band and self.position <= 0:
            # 价格触及下轨，买入（均值回归）
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='BUY',
                price=current_price,
                quantity=100,
                reason=f"价格触及布林带下轨: {current_price:.2f} <= {lower_band:.2f}"
            )
            self.position = 1
        
        elif current_price >= upper_band and self.position >= 0:
            # 价格触及上轨，卖出（均值回归）
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='SELL',
                price=current_price,
                quantity=100,
                reason=f"价格触及布林带上轨: {current_price:.2f} >= {upper_band:.2f}"
            )
            self.position = -1
        
        elif self.position == 1 and current_price >= middle_band:
            # 多头平仓：价格回到中轨
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='SELL',
                price=current_price,
                quantity=100,
                reason=f"价格回到布林带中轨: {current_price:.2f} >= {middle_band:.2f}"
            )
            self.position = 0
        
        elif self.position == -1 and current_price <= middle_band:
            # 空头平仓：价格回到中轨
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='BUY',
                price=current_price,
                quantity=100,
                reason=f"价格回到布林带中轨: {current_price:.2f} <= {middle_band:.2f}"
```

```python
class PairsTradingStrategy(BaseStrategy):
    """配对交易策略"""
    
    def __init__(self, symbol_a, symbol_b, lookback_period=60, entry_threshold=2, exit_threshold=0.5, name="Pairs_Trading"):
        super().__init__(name=name)
        self.symbol_a = symbol_a
        self.symbol_b = symbol_b
        self.lookback_period = lookback_period
        self.entry_threshold = entry_threshold
        self.exit_threshold = exit_threshold
        self.spread_mean = 0
        self.spread_std = 1
        self.position = 0  # 0: 无持仓, 1: 多A空B, -1: 空A多B
    
    async def on_data(self, data_a, data_b):
        """处理两只股票的数据"""
        if len(data_a) < self.lookback_period or len(data_b) < self.lookback_period:
            return
        
        # 计算价差
        price_a = data_a['close'].iloc[-self.lookback_period:]
        price_b = data_b['close'].iloc[-self.lookback_period:]
        
        # 计算价差（对数价格差）
        spread = np.log(price_a) - np.log(price_b)
        
        # 更新统计量
        self.spread_mean = spread.mean()
        self.spread_std = spread.std()
        
        current_spread = spread.iloc[-1]
        z_score = (current_spread - self.spread_mean) / self.spread_std
        
        current_price_a = price_a.iloc[-1]
        current_price_b = price_b.iloc[-1]
        
        # 生成交易信号
        if z_score > self.entry_threshold and self.position != -1:
            # 价差过大，做空A做多B
            self.generate_signal(
                symbol=self.symbol_a,
                signal_type='SELL',
                price=current_price_a,
                quantity=100,
                reason=f"配对交易：价差Z-score={z_score:.2f} > {self.entry_threshold}"
            )
            self.generate_signal(
                symbol=self.symbol_b,
                signal_type='BUY',
                price=current_price_b,
                quantity=100,
                reason=f"配对交易：对冲仓位"
            )
            self.position = -1
        
        elif z_score < -self.entry_threshold and self.position != 1:
            # 价差过小，做多A做空B
            self.generate_signal(
                symbol=self.symbol_a,
                signal_type='BUY',
                price=current_price_a,
                quantity=100,
                reason=f"配对交易：价差Z-score={z_score:.2f} < -{self.entry_threshold}"
            )
            self.generate_signal(
                symbol=self.symbol_b,
                signal_type='SELL',
                price=current_price_b,
                quantity=100,
                reason=f"配对交易：对冲仓位"
            )
            self.position = 1
        
        elif abs(z_score) < self.exit_threshold and self.position != 0:
            # 价差回归，平仓
            if self.position == 1:
                # 平多A空B
                self.generate_signal(
                    symbol=self.symbol_a,
                    signal_type='SELL',
                    price=current_price_a,
                    quantity=100,
                    reason=f"配对交易平仓：价差回归"
                )
                self.generate_signal(
                    symbol=self.symbol_b,
                    signal_type='BUY',
                    price=current_price_b,
                    quantity=100,
                    reason=f"配对交易平仓：对冲平仓"
                )
            else:
                # 平空A多B
                self.generate_signal(
                    symbol=self.symbol_a,
                    signal_type='BUY',
                    price=current_price_a,
                    quantity=100,
                    reason=f"配对交易平仓：价差回归"
                )
                self.generate_signal(
                    symbol=self.symbol_b,
                    signal_type='SELL',
                    price=current_price_b,
                    quantity=100,
                    reason=f"配对交易平仓：对冲平仓"
                )
            self.position = 0

# 使用示例
pairs_strategy = PairsTradingStrategy(
    symbol_a="US.AAPL",
    symbol_b="US.MSFT",
    lookback_period=60,
    entry_threshold=2,
    exit_threshold=0.5
)
```

```python
class BreakoutStrategy(BaseStrategy):
    """突破策略"""
    
    def __init__(self, lookback_period=20, breakout_multiplier=1.02, name="Breakout_Strategy"):
        super().__init__(name=name)
        self.lookback_period = lookback_period
        self.breakout_multiplier = breakout_multiplier
        self.position = 0
        self.resistance_level = None
        self.support_level = None
    
    async def on_data(self, data):
        """数据处理"""
        if len(data) < self.lookback_period:
            return
        
        # 计算阻力位和支撑位
        recent_high = data['high'].iloc[-self.lookback_period:].max()
        recent_low = data['low'].iloc[-self.lookback_period:].min()
        
        current_price = data['close'].iloc[-1]
        
        # 更新阻力位和支撑位
        self.resistance_level = recent_high * self.breakout_multiplier
        self.support_level = recent_low / self.breakout_multiplier
        
        # 生成交易信号
        if current_price >= self.resistance_level and self.position <= 0:
            # 向上突破，买入
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='BUY',
                price=current_price,
                quantity=100,
                reason=f"向上突破阻力位: {current_price:.2f} >= {self.resistance_level:.2f}"
            )
            self.position = 1
        
        elif current_price <= self.support_level and self.position >= 0:
            # 向下突破，卖出
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='SELL',
                price=current_price,
                quantity=100,
                reason=f"向下突破支撑位: {current_price:.2f} <= {self.support_level:.2f}"
            )
            self.position = -1
        
        # 止损逻辑
        elif self.position == 1 and current_price <= self.support_level:
            # 多头止损
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='SELL',
                price=current_price,
                quantity=100,
                reason=f"多头止损: 价格跌破支撑位"
            )
            self.position = 0
        
        elif self.position == -1 and current_price >= self.resistance_level:
            # 空头止损
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='BUY',
                price=current_price,
                quantity=100,
                reason=f"空头止损: 价格突破阻力位"
            )
            self.position = 0

# 使用示例
```

### 多因子选股策略

```python
from strategy.multi_factor.multi_factor_strategy import MultiFactorStrategy
from strategy.multi_factor.factor_combiner import FactorCombiner

class ValueGrowthStrategy(MultiFactorStrategy):
    """价值成长多因子策略"""
    
    def __init__(self, name="Value_Growth_Strategy"):
        super().__init__(name=name)
        
        # 定义因子权重
        self.factor_weights = {
            'value': 0.4,      # 价值因子
            'growth': 0.3,     # 成长因子
            'quality': 0.2,    # 质量因子
            'momentum': 0.1    # 动量因子
        }
    
    async def calculate_factors(self, universe_data):
        """计算各因子得分"""
        factors = {}
        
        # 价值因子（低市盈率、低市净率）
        if 'pe_ratio' in universe_data.columns and 'pb_ratio' in universe_data.columns:
            # 标准化处理
            pe_zscore = -self.zscore(universe_data['pe_ratio'])  # 负号表示低PE更好
            pb_zscore = -self.zscore(universe_data['pb_ratio'])
            factors['value'] = (pe_zscore + pb_zscore) / 2
        
        # 成长因子（营收增长率、盈利增长率）
        if 'revenue_growth' in universe_data.columns and 'earnings_growth' in universe_data.columns:
            revenue_growth_zscore = self.zscore(universe_data['revenue_growth'])
            earnings_growth_zscore = self.zscore(universe_data['earnings_growth'])
            factors['growth'] = (revenue_growth_zscore + earnings_growth_zscore) / 2
        
        # 质量因子（ROE、毛利率）
        if 'roe' in universe_data.columns and 'gross_margin' in universe_data.columns:
            roe_zscore = self.zscore(universe_data['roe'])
            gross_margin_zscore = self.zscore(universe_data['gross_margin'])
            factors['quality'] = (roe_zscore + gross_margin_zscore) / 2
        
        # 动量因子（价格动量）
        if 'price_momentum' in universe_data.columns:
            factors['momentum'] = self.zscore(universe_data['price_momentum'])
        
        return factors
    
    def zscore(self, series):
        """计算Z-score"""
        return (series - series.mean()) / series.std()
    
    async def generate_signals(self, universe_data):
        """生成交易信号"""
        # 计算因子得分
        factor_scores = await self.calculate_factors(universe_data)
        
        if not factor_scores:
            return
        
        # 组合因子得分
        combined_scores = FactorCombiner.combine_factors(
            factor_scores, 
            self.factor_weights
        )
        
        # 选择得分最高的股票
        top_stocks = combined_scores.nlargest(10)  # 选择前10只
        
        # 生成买入信号
        for symbol, score in top_stocks.items():
            current_price = universe_data.loc[symbol, 'close'] if 'close' in universe_data.columns else 100
            
            self.generate_signal(
                symbol=symbol,
                signal_type='BUY',
                price=current_price,
                weight=score / top_stocks.sum(),  # 按得分分配权重
                reason=f"多因子综合得分: {score:.4f}"
            )

# 使用示例
value_growth_strategy = ValueGrowthStrategy()
```

```python
        self.lookback_period = lookback_period
        self.prediction_horizon = prediction_horizon
        self.model = RandomForestRegressor(n_estimators=100, random_state=42)
        self.scaler = StandardScaler()
        self.is_trained = False
        self.position = 0
    
    def prepare_features(self, data):
        """准备特征数据"""
        features = []
        
        # 技术指标特征
        returns = data['close'].pct_change().dropna()
        volatility = returns.rolling(5).std()
        
        # 移动平均特征
        sma_5 = data['close'].rolling(5).mean()
        sma_10 = data['close'].rolling(10).mean()
        sma_20 = data['close'].rolling(20).mean()
        
        # 价格位置特征
        high_20 = data['high'].rolling(20).max()
        low_20 = data['low'].rolling(20).min()
        price_position = (data['close'] - low_20) / (high_20 - low_20)
        
        # 组合特征
        feature_df = pd.DataFrame({
            'return_1': returns,
            'return_5': returns.rolling(5).mean(),
            'volatility_5': volatility,
            'sma_5_ratio': data['close'] / sma_5,
            'sma_10_ratio': data['close'] / sma_10,
            'sma_20_ratio': data['close'] / sma_20,
            'price_position': price_position
        }).dropna()
        
        return feature_df
    
    def prepare_target(self, data, horizon):
        """准备目标变量（未来收益率）"""
        future_returns = data['close'].pct_change(horizon).shift(-horizon)
        return future_returns
    
    async def train_model(self, historical_data):
        """训练模型"""
        if len(historical_data) < self.lookback_period + self.prediction_horizon:
            print("数据不足，无法训练模型")
            return
        
        # 准备特征和目标
        features = self.prepare_features(historical_data)
        target = self.prepare_target(historical_data, self.prediction_horizon)
        
        # 对齐数据
        aligned_data = pd.concat([features, target], axis=1).dropna()
        
        if len(aligned_data) < 50:  # 最小训练样本数
            print("训练样本不足")
            return
        
        X = aligned_data.iloc[:, :-1]
        y = aligned_data.iloc[:, -1]
        
        # 标准化特征
        X_scaled = self.scaler.fit_transform(X)
        
        # 训练模型
        self.model.fit(X_scaled, y)
        self.is_trained = True
        
        print(f"模型训练完成，使用{len(aligned_data)}个样本")
    
    async def on_data(self, data):
        """数据处理"""
        if not self.is_trained or len(data) < self.lookback_period:
            return
        
        # 准备当前特征
        current_features = self.prepare_features(data.tail(self.lookback_period))
        
        if current_features.empty:
            return
        
        # 获取最新特征
        latest_features = current_features.iloc[-1:]
        
        # 标准化
        latest_features_scaled = self.scaler.transform(latest_features)
        
        # 预测
        prediction = self.model.predict(latest_features_scaled)[0]
        
        current_price = data['close'].iloc[-1]
        
        # 生成交易信号
        if prediction > 0.02 and self.position <= 0:  # 预测上涨超过2%
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='BUY',
                price=current_price,
                quantity=100,
                reason=f"机器学习预测上涨: {prediction:.4f}"
            )
            self.position = 1
        
        elif prediction < -0.02 and self.position >= 0:  # 预测下跌超过2%
            self.generate_signal(
                symbol=data['symbol'].iloc[-1] if 'symbol' in data.columns else 'default',
                signal_type='SELL',
                price=current_price,
                quantity=100,
                reason=f"机器学习预测下跌: {prediction:.4f}"
            )
            self.position = -1
```

```python
class RealtimePriceStrategy(BaseStrategy):
    """实时价格监控策略"""
    
    def __init__(self, price_change_threshold=0.05, volume_threshold=1.5, name="Realtime_Price_Strategy"):
        super().__init__(name=name)
        self.price_change_threshold = price_change_threshold  # 5%价格变动
        self.volume_threshold = volume_threshold  # 1.5倍成交量
        self.last_prices = {}
        self.average_volumes = {}
        self.volume_windows = {}
    
    async def on_realtime_data(self, data):
        """
        实时数据处理
        
        注意：data 参数现在是 pd.DataFrame 类型（单行），而非 dict
        """
        import pandas as pd
        
        if data is None or data.empty:
            return
        
        symbol = data['code'].iloc[0] if 'code' in data.columns else None
        data_type = data['data_type'].iloc[0] if 'data_type' in data.columns else None
        
        if data_type == 'quote':
            current_price = float(data['last_price'].iloc[0]) if 'last_price' in data.columns else None
            current_volume = int(data['volume'].iloc[0]) if 'volume' in data.columns else 0
            
            if symbol not in self.last_prices:
                # 初始化
                self.last_prices[symbol] = current_price
                self.average_volumes[symbol] = current_volume
                self.volume_windows[symbol] = [current_volume]
                return
            
            # 计算价格变动
            last_price = self.last_prices[symbol]
            price_change = (current_price - last_price) / last_price
            
            # 更新成交量窗口
            self.volume_windows[symbol].append(current_volume)
            if len(self.volume_windows[symbol]) > 20:  # 保持最近20个成交量
                self.volume_windows[symbol].pop(0)
            
            # 计算平均成交量
            avg_volume = np.mean(self.volume_windows[symbol])
            volume_ratio = current_volume / avg_volume if avg_volume > 0 else 1
            
            # 生成交易信号
            if abs(price_change) > self.price_change_threshold and volume_ratio > self.volume_threshold:
                signal_type = 'BUY' if price_change > 0 else 'SELL'
                
                self.generate_signal(
                    symbol=symbol,
                    signal_type=signal_type,
                    price=current_price,
                    quantity=100,
                    reason=f"价格变动{price_change:.2%}，成交量放大{volume_ratio:.1f}倍"
                )
            
            # 更新最后价格
            self.last_prices[symbol] = current_price
            self.average_volumes[symbol] = avg_volume
```

```python
class BacktestEngine:
    """简单回测引擎"""
    
    def __init__(self, initial_capital=100000):
        self.initial_capital = initial_capital
        self.capital = initial_capital
        self.positions = {}
        self.trades = []
        self.portfolio_values = []
    
    def run_backtest(self, strategy, historical_data):
        """运行回测"""
        self.capital = self.initial_capital
        self.positions = {}
        self.trades = []
        self.portfolio_values = []
        
        # 模拟逐日交易
        for i in range(len(historical_data)):
            current_data = historical_data.iloc[:i+1]
            current_price = historical_data['close'].iloc[i]
            
            # 调用策略
            # 这里需要异步处理，简化示例中直接调用
            # await strategy.on_data(current_data)
            
            # 更新投资组合价值
            portfolio_value = self.calculate_portfolio_value(current_price)
            self.portfolio_values.append(portfolio_value)
        
        return self.calculate_performance()
    
    def execute_trade(self, signal):
        """执行交易"""
        symbol = signal['symbol']
        signal_type = signal['signal_type']
        price = signal['price']
        quantity = signal.get('quantity', 100)
        
        trade_value = price * quantity
        
        if signal_type == 'BUY' and self.capital >= trade_value:
            # 买入
            if symbol not in self.positions:
                self.positions[symbol] = 0
            
            self.positions[symbol] += quantity
            self.capital -= trade_value
            
            self.trades.append({
                'symbol': symbol,
                'type': 'BUY',
                'price': price,
                'quantity': quantity,
                'timestamp': pd.Timestamp.now()
            })
        
        elif signal_type == 'SELL' and symbol in self.positions and self.positions[symbol] >= quantity:
            # 卖出
            self.positions[symbol] -= quantity
            self.capital += trade_value
            
            if self.positions[symbol] == 0:
                del self.positions[symbol]
            
            self.trades.append({
                'symbol': symbol,
                'type': 'SELL',
                'price': price,
                'quantity': quantity,
                'timestamp': pd.Timestamp.now()
            })
    
    def calculate_portfolio_value(self, current_price):
        """计算投资组合价值"""
        position_value = sum(
            quantity * current_price 
            for symbol, quantity in self.positions.items()
        )
        return self.capital + position_value
    
    def calculate_performance(self):
        """计算回测绩效"""
        if not self.portfolio_values:
            return {}
        
        final_value = self.portfolio_values[-1]
        total_return = (final_value - self.initial_capital) / self.initial_capital
        
        # 计算年化收益率
        days = len(self.portfolio_values)
        annual_return = (1 + total_return) ** (252 / days) - 1 if days > 0 else 0
        
        # 计算最大回撤
        peak = self.portfolio_values[0]
        max_drawdown = 0
        
        for value in self.portfolio_values:
            if value > peak:
                peak = value
            drawdown = (peak - value) / peak
            if drawdown > max_drawdown:
                max_drawdown = drawdown
        
        return {
            'initial_capital': self.initial_capital,
            'final_value': final_value,
            'total_return': total_return,
            'annual_return': annual_return,
            'max_drawdown': max_drawdown,
            'total_trades': len(self.trades),
            'win_rate': self.calculate_win_rate()
        }
    
    def calculate_win_rate(self):
        """计算胜率"""
        if len(self.trades) < 2:
            return 0
        
        # 简化计算，实际需要更复杂的逻辑
        return 0.5  # 示例值
```
