## 系统配置

### 基础配置

Easy-Quant使用统一的配置管理系统，配置文件位于 `utils/settings.py`：

```python
# utils/settings.py
from pydantic import BaseSettings
from typing import Optional

class DataSettings(BaseSettings):
    """数据相关配置"""
    data_dir: str = "data"
    cache_dir: str = "cache"
    max_cache_size: int = 1000
    cache_ttl: int = 3600  # 缓存过期时间（秒）

class LogSettings(BaseSettings):
    """日志相关配置"""
    log_level: str = "INFO"
    log_file: str = "logs/easy_quant.log"
    max_log_size: int = 100  # MB
    backup_count: int = 5

class SystemSettings(BaseSettings):
    """系统配置"""
    data: DataSettings = DataSettings()
    log: LogSettings = LogSettings()
    
    # 异步配置
    max_workers: int = 10
    timeout: int = 30  # 秒
    
    # 性能配置
    enable_cache: bool = True
    enable_compression: bool = True

# 全局配置实例
settings = SystemSettings()

def update_settings(**kwargs):
    """更新配置"""
    global settings
    
    # 更新数据配置
    if 'data_path' in kwargs:
        settings.data.data_dir = kwargs['data_path']
    if 'cache_path' in kwargs:
        settings.data.cache_dir = kwargs['cache_path']
    
    # 更新日志配置
    if 'log_level' in kwargs:
        settings.log.log_level = kwargs['log_level']
    if 'log_file' in kwargs:
        settings.log.log_file = kwargs['log_file']
    
    # 更新系统配置
    if 'max_workers' in kwargs:
        settings.max_workers = kwargs['max_workers']
    if 'timeout' in kwargs:
        settings.timeout = kwargs['timeout']
```

### 配置使用示例

```python
from utils.settings import settings, update_settings

# 查看当前配置
print(f"数据目录: {settings.data.data_dir}")
print(f"日志级别: {settings.log.log_level}")
print(f"最大工作线程数: {settings.max_workers}")

# 更新配置
update_settings(
    data_path="/path/to/your/data",
    log_level="DEBUG",
    max_workers=20
)

# 验证更新后的配置
print(f"更新后数据目录: {settings.data.data_dir}")
print(f"更新后日志级别: {settings.log.log_level}")
```

## 配置验证

### 配置验证工具

```python
from pydantic import BaseModel, ValidationError
from typing import Dict, Any

class ConfigValidator:
    """配置验证器"""
    
    @staticmethod
    def validate_data_config(config: Dict[str, Any]) -> bool:
        """验证数据配置"""
        required_fields = ['data_dir', 'cache_dir']
        
        for field in required_fields:
            if field not in config:
                raise ValueError(f"缺少必要字段: {field}")
        
        # 验证路径存在性
        if not os.path.exists(config['data_dir']):
            os.makedirs(config['data_dir'])
        
        if not os.path.exists(config['cache_dir']):
            os.makedirs(config['cache_dir'])
        
        return True
    
    @staticmethod
    def validate_strategy_config(config: Dict[str, Any]) -> bool:
        """验证策略配置"""
        # 验证数值范围
        if 'initial_capital' in config and config['initial_capital'] <= 0:
            raise ValueError("初始资金必须大于0")
        
        if 'commission_rate' in config and not (0 <= config['commission_rate'] <= 0.01):
            raise ValueError("佣金率必须在0-1%之间")
        
        return True
    
    @staticmethod
    def validate_all_configs(configs: Dict[str, Any]) -> Dict[str, bool]:
        """验证所有配置"""
        results = {}
        
        try:
            if 'data' in configs:
                results['data'] = ConfigValidator.validate_data_config(configs['data'])
            
            if 'strategy' in configs:
                results['strategy'] = ConfigValidator.validate_strategy_config(configs['strategy'])
            
            # 添加其他配置验证...
            
        except ValidationError as e:
            print(f"配置验证失败: {e}")
            return {key: False for key in results.keys()}
        
        return results

# 使用示例
configs = {
    'data': {
        'data_dir': 'data',
        'cache_dir': 'cache'
    },
    'strategy': {
        'initial_capital': 100000,
        'commission_rate': 0.0003
    }
}

validation_results = ConfigValidator.validate_all_configs(configs)
print(f"配置验证结果: {validation_results}")
```
